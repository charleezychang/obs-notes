#### Coverage Report
- coverage report - `ng test --no-watch --code-coverage` generates a report in `/coverage` directory after the tests are complete
- coverage could be configured to be generated every time you test via `angular.json`
- coverage could also be enforced (minimum requirement) via `coverageReporter` key in the configuration but may vary from different testing libraries
#### Testing Services
- `describe('testGroup', fn)` - group related tests together
- `beforeEach(fn)` - use to initialize services before test runs
- `it('testName', fn)` - for actual tests
```ts
// Straight Jasmine testing without Angular's testing support
  describe('ValueService', () => {
    let service: ValueService;
    beforeEach(() => {
      service = new ValueService();
    });
    it('#getValue should return real value', () => {
      expect(service.getValue()).toBe('real value');
    });
    it('#getObservableValue should return value from observable', (done: DoneFn) => {
      service.getObservableValue().subscribe((value) => {
        expect(value).toBe('observable value');
        done();
      });
    });
    it('#getPromiseValue should return value from a promise', (done: DoneFn) => {
      service.getPromiseValue().then((value) => {
        expect(value).toBe('promise value');
        done();
      });
    });
  });
```

##### Component Testing
###### Classic
```ts
it('should test Component', () => {
	const fixture = TestBed.configureTestingModule({
		declarations: [Component],
		imports: [
			AsyncPipe,
			NgIf,
			ReactiveFormsModule,
			MatButtonModule,
			MatFormFieldModule,
			MatInputModule,
			MatIconModule,
			HttpClientTestingModule,
			NoopAnimationsModule,
		]
	}).createComponent(Component)
})
```
But since Component is a standalone, it also acts as its own module, so you can simplify the testing module:
```ts
it('should test Component', fakeAsync(() => {
	const fixture = TestBed.configureTestingModule({
		imports: [
			Component,
			HttpClientTestingModule,
			NoopAnimationsModule,
		]
	}).createComponent(Component)
	// make sures directives are executive, inputs, etc.
	fixture.detectChanges();
	// data-testid is an attribute specifically for testing purposes
	// <input data-testid="address" formControlName="address" />
	const input = fixture.debugElement.query(By.css("[data-testid=address]")).nativeElement as HTMLInputElement
	input.value = "Domgasse 5";
	// issue: reactive forms is expecting a DOM event but value was changed programatically
    // to do this manually...
    input.dispatchEvent(new Event("input"))
    fixture.debugElement.query(By.css("[data-testid=btn-search]")).nativeElement.click()
    // issue: asynchrony from http request
    TestBed.inject(HttpTestingController).expectOne(req => !!req.url.match(/nominatim/)).flush([true])
    // with test bed, it is tested synchronously, thats is why the test is wrapped with fakeAsync which exposes tick()
    tick();
    // re-render the HTML element that uses fetch data
    fixture.detectChanges();
    const message = fixture.debugElement.query(By.css("[data-testid=lookup-result]")).nativeElement as HTMLParagraphElement;
    // always induce failure into tests first to check if test is actually working (message should be 'Brochure sent')
    expect(message.textContent).toBe('Brochure send')
}))
```

#### Playwright
https://www.youtube.com/watch?v=9SHrznGRPQA
- by Microsoft
- no dependencies
- acts like remote control for the browser
- runs default on 3 projects: chromium, firefox, webkit (safari)

Installation and Configuration
- `npm init playwright`
- *playwright.config.ts* - modify base URL, recommended to enable `webServer`, `fullyParallel: true`
- recommended: *package.json* `"scripts" : { "e2e:dev": "playwright test --ui" }`
- in default test folder, remove auto-generated test file and make own tests (file extension: *.spec.ts*)
```ts
webServer: {
	command: 'npm run start',
	url: 'http://localhost:4200',
	reuseExistingServer: !process.env.CI
}
```

Writing Tests
```ts
test.describe('test suite name', () => {
	// Library exposes page object which lets you control browser
	test('test case name', async ({page: Page}) => {
        // Almost all playwright methods are async so use await keyword
        // Base Url is already defined in config so goto('') is already pointed to that
		await page.goto('')
        // But PW will load the page but not fully render the elements
        // By using locator, PW will wait the page to render until the selector is found
        // This uses CSS selector, but its recommended to use data-testid attribute
        await page.locator('[data-testid=btn-admin]').click()
        // Then verify/assert if the event had happened (assertion)
        await expect(page.locator('[data-testid=row-holiday]')).toHaveCount(2);
        // Alternatively: await page.getByTestId('btn-admin').click();
        // Aside via CSS Selector, you can use the User Facing selector in the PW UI as a locator
        await page.getByRole('link', { name: 'Add Holiday' }).click();
        // Method fill() clears the content and overwrites
        await page.getByLabel('Name').fill('Austria');
        await page.getByLabel('Description').fill('Austria has mountains');
        await page.getByRole('button', { name: 'Save' }).click();
        // ALWAYS LET YOUR TESTS FAIL FIRST
        // await expect(page.locator('[data-testid=row-holiday]')).toHaveCount(2);
        await expect(page.locator('[data-testid=row-holiday]')).toHaveCount(3);
	})
})
```

VSCode Integration
- will add play button beside `test()` methods lines
- able to add breakpoints
- will start browser, you can interact with the elements unlike with PW UI
- allows highlighting in the browser while hovering over locators
- also allows modification of test cases even during testing

Browser Developer Tools
- requires browser extension
- able to record steps using internal recorder of browser and export as PW code
